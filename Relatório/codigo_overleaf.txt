\section{Requisitos do Processador}
\begin{itemize}
    \item Palavra de pelo menos 8 bits.
    \item Registradores obrigatórios: 3 para operações matemáticas e 1 para flags.
    \item Conjunto mínimo de instruções:
        \begin{itemize}
            \item Movimentação e transferência de dados
            \item Manipulação de bits e deslocamento
            \item Operações aritméticas: ADD, SUB, MUL, DIV, MOD
            \item Operações relacionais: comparação
            \item Operações lógicas: AND, OR, NOT, XOR
            \item Outros: macros, constantes, operações I/O, NOP, HALT
        \end{itemize}
\end{itemize}
